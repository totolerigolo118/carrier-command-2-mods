/*
 * Patch "cancel the launch of an aircraft already picked up by the crane" for Carrier Command 2
 * (build_mac of 2026-01-05, md5 b0107170...). public tracker request no. 20652, study part 02 of the notes, section 5.
 *
 * Compiled by generer_grue_annulation.py (clang -arch x86_64), placed in the dead code of
 * ImGui::ShowDemoWindow (never called by the game). Two hooks:
 *   A 0x1001a8c01  docking_manager_air::undock, aircraft already in the crane/lift/runway: the game only does
 *                  something if the aircraft is waiting at launch (dock_state 7); otherwise we call ag_annuler.
 *   B 0x1001ad9ad  docking_manager_air::update, launch mode, just before "end of crane -> lift":
 *                  ag_fin releases the crane when an aircraft brought back to its bay has been set down.
 *
 * Aircraft manager (carrier + 0x15e8): 0x00 mode (2 = launch), 0x08 list of crane moves (ring buffer of
 * ints: +0 pointer, +8 capacity, +0xc head, +0x10 count), 0x20 the crane is holding the aircraft,
 * 0x78 id of the crane's aircraft, 0x88 its bay. Moves: 1 go to the bay, 2 return above the lift,
 * 3 rotate, 4 grab then raise (when already holding: raise), 5 set down then raise (when holding
 * nothing: raise). Carrier: 0x1548 crane rotation (0 or pi).
 * Aircraft: 0x8c id, 0x220 dock_state (3 launching, 4 in the hangar), 0x21c launch phase (0 in the hangar).
 * While being carried, the aircraft stays stowed in its bay: only the bay position follows the hook.
 */

typedef unsigned int u32;
typedef unsigned char u8;

#define AT(p, o)    ((char *)(p) + (o))
#define U32(p, o)   (*(u32 *)AT(p, o))
#define U8(p, o)    (*(u8 *)AT(p, o))
#define F32(p, o)   (*(float *)AT(p, o))
#define PTR(p, o)   (*(void **)AT(p, o))
#define VFN(obj, off) ((*(void ***)(obj))[(off) / 8])

#define HIDDEN __attribute__((visibility("hidden")))

#define AUCUN       0xffffffffu
#define OFF_CARRIER 0x15e8        /* the aircraft manager is embedded in the carrier */
#define ROT_MAX     0.01f         /* rad: beyond this, the aircraft has started to be rotated -> refuse */

/* game functions (addresses in the generator's JEU table) */
extern u32 *g_push_geste(void *liste);                      /* mm_vector<e_crane_state>::push_back() */
extern void g_vider_route(void *route);                     /* mm_vector<c_vehicle_waypoint_data>::clear() */
extern void g_reset_route(void *veh);                       /* c_vehicle_server::reset_waypoint_path() */

typedef void (*fn_carte)(void *, u32, void *, void *, u32);

static u32 anneau(u32 tete, u32 i, u32 cap)
{
    u32 k = tete + i;
    return k < cap ? k : k - cap;
}

static void *vehicule(void *vcont, u32 id)
{
    u32 n = U32(vcont, 0x60), cap = U32(vcont, 0x58), tete = U32(vcont, 0x5c);
    void **tab = (void **)PTR(vcont, 0x50);
    for (u32 i = 0; i < n; i++) {
        void *v = tab[anneau(tete, i, cap)];
        if (v && U32(v, 0x8c) == id)
            return v;
    }
    return 0;
}

static int geste_present(void *mgr, u32 geste)
{
    u32 *tab = (u32 *)PTR(mgr, 0x08);
    u32 cap = U32(mgr, 0x10), tete = U32(mgr, 0x14), n = U32(mgr, 0x18);
    for (u32 i = 0; i < n; i++)
        if (tab[anneau(tete, i, cap)] == geste)
            return 1;
    return 0;
}

static void nouvelle_liste(void *mgr, const u32 *gestes, u32 n)
{
    U32(mgr, 0x14) = 0;
    U32(mgr, 0x18) = 0;
    for (u32 i = 0; i < n; i++)
        *g_push_geste(AT(mgr, 0x08)) = gestes[i];
}

/* 2nd press of the bay button while the crane has already picked up the aircraft (launch in progress).
   Returns 1 if the launch is cancelled, 0 if the game is left to proceed (too late or other case). */
HIDDEN int ag_annuler(void *mgr, void *veh, void *carte)
{
    static const u32 sans_avion[] = { 5, 2 };          /* raise, return to rest */
    static const u32 avec_avion[] = { 4, 1, 5, 2 };    /* raise, go to the bay, set down, return */
    void *carrier = AT(mgr, -OFF_CARRIER);
    u32 id = U32(veh, 0x8c);
    float rot;

    if (U32(mgr, 0x00) != 2 || U32(mgr, 0x78) != id || U32(veh, 0x220) != 3)
        return 0;
    if (!U8(mgr, 0x20)) {
        if (!geste_present(mgr, 4))        /* already released above the lift */
            return 0;
        nouvelle_liste(mgr, sans_avion, 2);
        U32(mgr, 0x78) = AUCUN;            /* the aircraft has not moved: the crane is free immediately */
        U32(mgr, 0x88) = AUCUN;
    } else {
        rot = F32(carrier, 0x1548);
        if (rot > ROT_MAX || rot < -ROT_MAX)   /* already being turned around above the lift */
            return 0;
        nouvelle_liste(mgr, avec_avion, 4);    /* 0x78 kept: the aircraft follows the hook to its bay */
    }
    U32(veh, 0x220) = 4;
    U32(veh, 0x21c) = 0;
    /* as in docking_manager_air::cancel_undocking: route cleared, map notified */
    g_vider_route(AT(veh, 0x1d0));
    g_reset_route(veh);
    ((fn_carte)VFN(carte, 0))(carte, id, AT(veh, 0x300), AT(veh, 0x1d0), U32(veh, 0x1e8));
    return 1;
}

/* Launch mode, on every tick: when the crane has finished setting down a returned aircraft (dock_state 4
   in the crane, which the original game never does in launch mode), it is released instead of handing the
   aircraft over to the lift. State read from the saved game: an interrupted return resumes on load. */
HIDDEN void ag_fin(void *mgr, void *vcont)
{
    void *v;
    u32 id = U32(mgr, 0x78);

    if (U32(mgr, 0x00) != 2 || U32(mgr, 0x18) != 0 || id == AUCUN || U8(mgr, 0x20))
        return;
    v = vehicule(vcont, id);
    if (v && U32(v, 0x220) == 4) {
        U32(mgr, 0x78) = AUCUN;
        U32(mgr, 0x88) = AUCUN;
    }
}

/* A: game registers at 0x1001a8c01: rdi manager, r14 aircraft, ebx its id, rcx map; rsp 16-byte aligned;
   returns to the epilogue 0x1001a8c0e, which reads no register.
   B: at 0x1001ad9ad, r14 manager, vehicle container at -0x38(%rbp), rsp 16-byte aligned; landing point
   of 4 game jumps: every register is saved, then cmpl $0,0x18(%r14) is replayed. */
__asm__(
    ".text\n"
    ".p2align 4\n"
    ".globl _ag_a\n"
    "_ag_a:\n"
    "    cmpl $7, 0x220(%r14)\n"
    "    jne 1f\n"
    "    movl %ebx, 0x28(%rdi)\n"
    "    jmp g_ret_a8c0e\n"
    "1:\n"
    "    movq %r14, %rsi\n"
    "    movq %rcx, %rdx\n"
    "    call _ag_annuler\n"
    "    jmp g_ret_a8c0e\n"
    "\n"
    ".p2align 4\n"
    ".globl _ag_b\n"
    "_ag_b:\n"
    "    pushq %rax\n pushq %rcx\n pushq %rdx\n pushq %rsi\n pushq %rdi\n"
    "    pushq %r8\n pushq %r9\n pushq %r10\n pushq %r11\n"
    "    subq $264, %rsp\n"
    "    movdqu %xmm0, 0(%rsp)\n movdqu %xmm1, 16(%rsp)\n movdqu %xmm2, 32(%rsp)\n movdqu %xmm3, 48(%rsp)\n"

    "    popq %r11\n popq %r10\n popq %r9\n popq %r8\n"
    "    popq %rdi\n popq %rsi\n popq %rdx\n popq %rcx\n popq %rax\n"
    "    cmpl $0, 0x18(%r14)\n"
    "    jmp g_ret_ad9b2\n"
);

/* Patch "squadron speed" for Carrier Command 2 (build_mac of 2026-01-05, md5 b0107170...).
 *
 * Study: part 05 of the notes. Feature: on the vehicle control screen, choose the cruise speed of the
 * player's own AI-controlled aircraft (ECO / NORMAL / FLANK).
 *
 * 1. _vit_plancher (assembly, further down) replaces 0x1001c2b4b..0x1001c2b8e of
 *    c_vehicle_server_wing::tick_control: when flying to a point (control_mode 0), the target speed there is
 *    clamped to [48, 144] (Albatros) or [72, 144] (Manta, type 10). Here the floor depends on the mode chosen
 *    for the aircraft's team (table g_vit_modes); an AI-controlled team (is_ai_controlled), no mode or an
 *    unknown mode = original game behaviour. Registers xmm0/xmm1 and the field at 0x2c8 end up exactly as
 *    in the original game.
 * 2. vit_lua is a Lua function "cc2_vitesse" (it takes the place of dofile in base_funcs, which neither the
 *    game nor mods use):
 *      cc2_vitesse(team)        -> the team's mode (0 = not chosen yet)
 *      cc2_vitesse(team, mode)  -> sets the mode (1 ECO, 2 NORMAL, 3 FLANK), returns it
 *    The last mode chosen is kept in a small file (CHEMIN) and restored on the first call of a later run.
 *
 * Writable memory: end of the ShowDemoWindowWidgets()::text buffer (0x1029e5680..0x1029e56a0, __data,
 * zeros in the file), which only the dead code of the ImGui demo reads.
 */

#ifndef CHEMIN
#error "CHEMIN must be provided by the generator (-DCHEMIN=...)"
#endif

typedef struct lua_State lua_State;
typedef struct FILE FILE;

extern int g_lua_gettop(lua_State *L);
extern long long g_lua_tointegerx(lua_State *L, int idx, int *isnum);
extern void g_lua_pushinteger(lua_State *L, long long n);
extern FILE *g_fopen(const char *chemin, const char *mode);
extern int g_fgetc(FILE *f);
extern int g_fputs(const char *s, FILE *f);
extern int g_fclose(FILE *f);

/* modes per team number (0..15); etat[0] = file already read, etat[1] = mode stored in the file */
extern unsigned char g_vit_modes[16] __attribute__((visibility("hidden")));
extern unsigned char g_vit_etat[2] __attribute__((visibility("hidden")));

#define NB_EQUIPES 16
#define MODE_MAX 3

static const char chemin[] = CHEMIN;
static const char lecture[] = "r";
static const char ecriture[] = "w";

static int lire_fichier(void)
{
    FILE *f = g_fopen(chemin, lecture);
    if (!f)
        return 0;
    int c = g_fgetc(f);
    g_fclose(f);
    return (c >= '1' && c <= '0' + MODE_MAX) ? c - '0' : 0;
}

static void ecrire_fichier(int mode)
{
    FILE *f = g_fopen(chemin, ecriture);
    if (!f)
        return;
    char texte[3] = { (char)('0' + mode), '\n', 0 };
    g_fputs(texte, f);
    g_fclose(f);
}

int vit_lua(lua_State *L)
{
    int n = g_lua_gettop(L);
    long long equipe = g_lua_tointegerx(L, 1, 0);
    long long mode = 0;
    if (equipe >= 0 && equipe < NB_EQUIPES) {
        if (!g_vit_etat[0]) {
            g_vit_etat[0] = 1;
            g_vit_etat[1] = (unsigned char)lire_fichier();
        }
        if (n >= 2) {
            long long m = g_lua_tointegerx(L, 2, 0);
            if (m >= 1 && m <= MODE_MAX) {
                g_vit_modes[equipe] = (unsigned char)m;
                if (m != g_vit_etat[1]) {
                    ecrire_fichier((int)m);
                    g_vit_etat[1] = (unsigned char)m;
                }
            }
        } else if (g_vit_modes[equipe] == 0) {
            g_vit_modes[equipe] = g_vit_etat[1];
        }
        mode = g_vit_modes[equipe];
    }
    g_lua_pushinteger(L, mode);
    return 1;
}

/* Lua name of the function (pointed to by the "dofile" entry of base_funcs). */
const char vit_nom_lua[] = "cc2_vitesse";

/* Speed floors (m/s) per mode: { Albatros (and any other aircraft), Manta }.
 * 0 = no mode and 1 = ECO: original game. NORMAL = estimate of about 80 % power (study,
 * section 2), to be tuned after in-game testing. FLANK = the ceiling: the throttle stays at maximum. */
const float vit_planchers[MODE_MAX + 1][2] = {
    { 48.0f, 72.0f },
    { 48.0f, 72.0f },
    { 61.0f, 91.0f },
    { 144.0f, 144.0f },
};
const float vit_plafond = 144.0f;

/* Hook at 0x1001c2b4b (67 bytes replaced). Entry: rbx -> speed returned by
 * get_waypoint_position_direction_speed, r13 = the aircraft. rax and rcx are saved; flags are free
 * (the game recomputes them at 0x1001c2ba7 before reading them). No call: stack intact. */
__asm__(
    ".text\n"
    ".globl _vit_plancher\n"
    "_vit_plancher:\n"
    "    movss (%rbx), %xmm0\n"                 /* replays 0x1001c2b4b */
    "    pushq %rax\n"
    "    pushq %rcx\n"
    "    xorl %ecx, %ecx\n"                     /* mode 0 = original game */
    "    movq 0x260(%r13), %rax\n"              /* the aircraft's team (c_team_server*) */
    "    testq %rax, %rax\n"
    "    je 1f\n"
    "    cmpb $0, 0x19(%rax)\n"                 /* is_ai_controlled: enemy aircraft unchanged */
    "    jne 1f\n"
    "    movl 0x14(%rax), %eax\n"               /* team number */
    "    cmpl $16, %eax\n"
    "    jae 1f\n"
    "    leaq _g_vit_modes(%rip), %rcx\n"
    "    movzbl (%rcx,%rax), %ecx\n"
    "    cmpl $3, %ecx\n"
    "    jbe 1f\n"
    "    xorl %ecx, %ecx\n"
    "1:\n"
    "    leaq _vit_planchers(%rip), %rax\n"
    "    movaps %xmm0, %xmm1\n"
    "    maxss (%rax,%rcx,8), %xmm1\n"           /* 0x1001c2b52 : maxss 48 */
    "    minss _vit_plafond(%rip), %xmm1\n"     /* 0x1001c2b5a : minss 144 */
    "    movss %xmm1, 0x2c8(%r13)\n"
    "    cmpl $0xa, 0x90(%r13)\n"               /* Manta? */
    "    jne 2f\n"
    "    maxss 4(%rax,%rcx,8), %xmm0\n"          /* 0x1001c2b75 : maxss 72 */
    "    minss _vit_plafond(%rip), %xmm0\n"
    "    movss %xmm0, 0x2c8(%r13)\n"
    "2:\n"
    "    popq %rcx\n"
    "    popq %rax\n"
    "    jmp _g_ret_2b8e\n"                     /* game continues: movl 0x318(%r13), %esi */
);

# Patches "sharper CCTV screen" for Carrier Command 2 (build_mac of 2026-01-05, md5 b0107170...).
#
# The image of a bridge screen serves both as the background (the camera view) and as the Lua script's interface.
# Its size comes from a per-screen-type table (type 0 = 128 x 128, 35 screens). Here, only the screen whose script
# is "scripts/screen_cctv.lua" gets a TAILLE x TAILLE image; the interface keeps its 128 units.
#   A  create_lite 0x1000ad74f  : image size at creation (tables 0x1028b1470 / 0x1028b1488)
#   B  render_screen 0x10021ce70 : interface camera = 128 x 128 if the image is TAILLE x TAILLE,
#                                  and scale = TAILLE / 128 in g_echelle (0 = no scale)
#   C  element_clip 0x10015e662  : glScissor(x, y, w, h) multiplied by the scale
#   D  render_screen 0x10021cf13 : scale reset to 0 after the interface (piloting, pause, overlay underneath)
# g_echelle: ShowExampleAppSimpleOverlay()::corner (0x102a2a864, __bss, 4 bytes), read only by the dead code.
# Registers: only those that the original code overwrites right after (A: rax rcx rdx r8 r9 ; B: rcx xmm0 xmm2 ;
# C: xmm0-xmm3 ecx edx esi edi ; D: edi).
# TAILLE and ECHELLE (bits of the float TAILLE / 128) are defined by the generator.

.text
.globl _cctv_a
_cctv_a:                                # rax = screen definitions, rcx = index x 0x78
    leaq (%rax,%rcx), %rdx
    movslq 0x60(%rdx), %rax             # screen type (original instruction)
    cmpl $23, 0x20(%rdx)                # length of script_file_name (mm_string: text 0x18, length 0x20)
    jne 1f
    movq 0x18(%rdx), %rcx
    testq %rcx, %rcx
    je 1f
    movq (%rcx), %r8
    cmpq L_script(%rip), %r8
    jne 1f
    movq 8(%rcx), %r8
    cmpq L_script+8(%rip), %r8
    jne 1f
    movq 15(%rcx), %r8
    cmpq L_script+15(%rip), %r8
    jne 1f
    movl $TAILLE, %r9d
    movl $TAILLE, %r8d
    jmp g_ret_a
1:                                      # original code
    movl $0x80, %r9d
    movl $0x80, %r8d
    cmpq $5, %rax
    ja 2f
    leaq g_tab_haut(%rip), %rcx
    movl (%rcx,%rax,4), %r9d
    leaq g_tab_larg(%rip), %rcx
    movl (%rcx,%rax,4), %r8d
2:
    jmp g_ret_a

.globl _cctv_b
_cctv_b:                                # r14 = screen image (width 0x20, height 0x24)
    movl 0x20(%r14), %ecx
    cmpl $TAILLE, %ecx
    jne 1f
    cmpl $TAILLE, 0x24(%r14)
    jne 1f
    movss g_128(%rip), %xmm0
    movss g_128(%rip), %xmm2
    movl $ECHELLE, g_echelle(%rip)
    jmp g_ret_b
1:                                      # original code
    xorps %xmm0, %xmm0
    cvtsi2ss %rcx, %xmm0
    movl 0x24(%r14), %ecx
    cvtsi2ss %rcx, %xmm2
    movl $0, g_echelle(%rip)
    jmp g_ret_b

.globl _cctv_c
_cctv_c:                                # rdi = element : x 0x8, y 0xc, w 0x10, h 0x14
    movss g_echelle(%rip), %xmm2
    xorps %xmm3, %xmm3
    ucomiss %xmm3, %xmm2
    jne 1f
    movss g_demi(%rip), %xmm0           # original code
    movss 0x14(%rdi), %xmm1
    addss %xmm0, %xmm1
    roundss $9, %xmm1, %xmm1
    cvttss2si %xmm1, %ecx
    addss 0x10(%rdi), %xmm0
    roundss $9, %xmm0, %xmm0
    cvttss2si %xmm0, %edx
    xorps %xmm0, %xmm0
    roundss $9, 0xc(%rdi), %xmm0
    cvttss2si %xmm0, %esi
    xorps %xmm0, %xmm0
    roundss $9, 0x8(%rdi), %xmm0
    cvttss2si %xmm0, %edi
    jmp g_ret_c
1:                                      # same computation, interface units x scale
    movss 0x14(%rdi), %xmm1
    mulss %xmm2, %xmm1
    addss g_demi(%rip), %xmm1
    roundss $9, %xmm1, %xmm1
    cvttss2si %xmm1, %ecx
    movss 0x10(%rdi), %xmm0
    mulss %xmm2, %xmm0
    addss g_demi(%rip), %xmm0
    roundss $9, %xmm0, %xmm0
    cvttss2si %xmm0, %edx
    movss 0xc(%rdi), %xmm0

#
# Lua's print() (luaB_print, 0x10283ae97) writes to standard output, which the game, when launched by Steam,
# throws away. Two instructions of luaB_print read the address of __stdoutp from the GOT:
#   0x10283aed0  movq GOT(__stdoutp), %r12   (before the loop that writes the arguments)
#   0x10283af78  movq GOT(__stdoutp), %rbx   (before the final "\n" and fflush; also the print() without arguments path)
# They become "call _rv_r12" and "call _rv_rbx": the register receives the address of a cell that holds
# a FILE* opened once and for all (fopen "a" on the log file), or stdout if the open fails.
# Cell: guard variable of ShowExampleAppLog()::log (0x102a2a828, 8 bytes in __bss), which only the dead code
# of ImGui::ShowDemoWindow uses.
# Stack: in luaB_print, rsp = 0 mod 16; _rv_xxx is entered at 8, _rv_ouvrir at 0: the call to fopen is aligned.
# Only registers that are free at this point of luaB_print are modified (rax, rcx, rdx, rsi, rdi, r8-r11).
# __CHEMIN__ is replaced by the generator (the test replaces it with a temporary file).

.globl _rv_r12
_rv_r12:
    call _rv_ouvrir
    leaq _g_case(%rip), %r12
    ret

.globl _rv_rbx
_rv_rbx:
    call _rv_ouvrir
    leaq _g_case(%rip), %rbx
    ret

_rv_ouvrir:
    movq _g_case(%rip), %rax
    testq %rax, %rax
    jne 1f
    leaq L_chemin(%rip), %rdi
    leaq L_mode(%rip), %rsi
    call _g_fopen
    testq %rax, %rax
    jne 2f
    movq _g_got_stdoutp(%rip), %rax
    movq (%rax), %rax

/*
 * Patch "helicopters set down on the lift" for Carrier Command 2 (build_mac of 2026-01-05, md5 b0107170...).
 *
 * Compiled by generer_helicos_ascenseur.py (clang -arch x86_64), then placed in the dead code of
 * ImGui::ShowDemoWindow (0x10005af39..0x100061c00, never called by the game). Three hooks in
 * c_vehicle_server_carrier::docking_manager_air::update (0x1001accd8):
 *   H1 0x1001acd94  entry of the "returns" mode: calls hl_tick on every tick;
 *   H2 0x1001ad7af  exit of the "returns" mode: forbidden while the lift lane is occupied;
 *   H3 0x1001ae071  taxiing state 6 -> 7 (towards the lift): waits if a helicopter is descending onto the platform.
 *
 * Lift lane = field 0x84 of the manager (vehicle_id_rotor_takeoff, saved by the game, unused during returns).
 * One helicopter at a time; the other aircraft land on the runway as before.
 * Lane state in two free fields of the manager (padding, never read by the game):
 *   0x9c  0xA5 << 24 | engaged << 23 | tick counter (23 bits)
 *   0x114 0xB6 << 24 | id of the last helicopter sent back to the runway (24 bits)
 * Game reference frame: y up, mat34_f64 matrices = 9 doubles of rotation followed by the translation (+0x48).
 * All matrix x vector multiplications go through the game's function (same convention).
 */

static double rac(double x) { return __builtin_sqrt(x > 0.0 ? x : 0.0); }
static double borne(double x, double a, double b) { return x < a ? a : (x > b ? b : x); }

static u32 anneau(u32 tete, u32 i, u32 cap)
{
    u32 k = tete + i;
    return k < cap ? k : k - cap;
}

static void *vehicule(void *vcont, u32 id)

        void *v = tab[anneau(tete, i, cap)];

    return 0;
}

static int joueurs_a_bord(void *carrier, void *peers)

static u32 etat_lire(void *mgr)

/* failure: like a failed approach in the game, the helicopter goes back into the queue and will use the runway */
static void renvoyer_vers_piste(void *mgr, void *veh)
{

    reglages_vol(veh, 0.0f, 1.0f);
    F32(veh, 0x2cc) = 1.0f;

    el[0] = id;
    el[1] = 3;
    el[2] = 0;

    train(veh, 0);
    U32(mgr, 0x114) = (TAG_REFUS << 24) | (id & 0xffffffu);

}

/* the Petrel drops its vehicle into the hangar, as during the game's approach */
static void lacher_cargaison(void *carrier, void *veh, void *vcont, void *map, void *scene)
{
    u32 n = U32(veh, 0x1f8);
    for (u32 i = 0; i < n; i++) {
        u32 id = U32(AT(PTR(veh, 0x1f0), i * 0x70), 0x68);
        if (!id)
            continue;
        void *cargo = vehicule(vcont, id);
        if (!cargo)
            continue;
        g_undock_immediate(veh, map, scene, cargo);
        u32 baie = g_bay_index_type(carrier, U32(cargo, 0x90), U8(cargo, 0x234));
        if (baie < U32(carrier, 0x1f8))
            g_dock_immediate(carrier, map, cargo, baie);
    }
}

/* set down on the platform: attachment identical to the game's touchdown (0x1001b2061), then hand-over to
   the lift as at the end of taxiing state 7 (0x1001ae13a) */
static int poser(void *mgr, void *carrier, void *veh, void *map, void *peers, void *scene)
{
    void *def = PTR(carrier, 0x98);
    u32 baie = g_bay_index_attach(carrier, 1, U8(veh, 0x234));
    if (baie >= U32(def, 0x140))
        return 0;
    u32 vid = U32(veh, 0x8c), cid = U32(carrier, 0x8c);

    reglages_vol(veh, 0.0f, 1.0f);
    F32(veh, 0x2cc) = 0.0f;
    void *att = AT(PTR(carrier, 0x1f0), baie * 0x70);
    g_set_attached_to(veh, map, cid);
    U64(veh, 0x198) = 0;
    I32(veh, 0x1a0) = -1;
    U8(veh, 0x234) = (u8)baie;
    double m[12];
    matrice_ascenseur(carrier, mgr, m);
    if (U32(att, 0x68) != vid)
        U32(att, 0x68) = vid;
    g_set_att_transform(att, m);
    g_create_physics(att, carrier, veh);
    void *joint = PTR(att, 0x60);
    *g_push_joint(AT(scene, 0x60)) = joint;
    ((fn_void)VFN(joint, 0x10))(joint);
    ((fn_map_dock)VFN(map, 0x40))(map, cid, baie, vid);
    U64(veh, 0x21c) = 0x600000001ull;
    u32 nr = U32(veh, 0x208);
    void **roues = (void **)PTR(veh, 0x200);
    for (u32 i = 0; i < nr; i++)
        if (roues[i])
            U8(roues[i], 0x134) = 0;
    u32 np = U32(peers, 0x10), cap = U32(peers, 0x8), tete = U32(peers, 0xc);
    void **pairs = (void **)PTR(peers, 0);
    for (u32 i = 0; i < np; i++) {
        void *pair = pairs[anneau(tete, i, cap)];
        void *ev = g_new(0x10);
        PTR(ev, 0) = g_event_vtable;
        *g_push_event(AT(PTR(pair, 0x68), 0x10)) = ev;
        U32(ev, 8) = vid;
        U16(ev, 0xc) = (u16)cid;
    }

    /* the lift takes the helicopter; the game lowers it and taxis it to the crane */
    I32(mgr, 0x7c) = (int)vid;
    I32(mgr, 0x8c) = (int)baie;
    if (U32(mgr, 0xd8)) {
        g_free(PTR(mgr, 0xd0));
        PTR(mgr, 0xd0) = 0;
        U32(mgr, 0xd8) = 0;
    }
    U64(mgr, 0xdc) = 0;
    U64(mgr, 0xe8) = 0;
    u64 chaine[2];
    g_mm_string_ctor(chaine, g_str_load_lift_lower);
    void *lien = g_get_link(def, chaine);
    if ((void *)chaine[0]) {
        g_free((void *)chaine[0]);
        chaine[1] = 0;
    }
    g_path_from_link_rev(AT(mgr, 0xd0), def, lien);
    liberer(mgr);
    return 1;
}

/* 1 = all wheels in contact (same test as the game), 0 = no, -1 = no wheel known */
static int roues_au_contact(void *veh)
{
    u32 n = U32(veh, 0xd8);
    if (!n)
        return -1;
    char *r = (char *)PTR(veh, 0xd0);
    for (u32 i = 0; i < n; i++) {
        float compression = F32(r, i * 0x70), glisse = F32(r, i * 0x70 + 4);
        if (compression < 0.01f || glisse > 0.1f)
            return 0;
    }
    return 1;
}

static void piloter(void *mgr, void *carrier, void *veh, void *vcont, void *map, void *peers, void *scene)
{
    u32 e = etat_lire(mgr);
    u32 ticks = (e & 0x7fffffu) + 1;
    int engage = (e & ENGAGE) != 0;

    double *c = corps0(carrier);
    double *p = corps0(veh);
    double nm[12];
    matrice_ascenseur(carrier, mgr, nm);
    vec3d haut = vers_monde(c, (vec3d){ nm[9], nm[10], nm[11] });
    vec3d cap_ascenseur = tourner(c, tourner(nm, (vec3d){ 0.0, 0.0, 1.0 }));
    vec3d cap_bateau = tourner(c, (vec3d){ 0.0, 0.0, 1.0 });
    vec3f vb = vitesse(carrier), vh = vitesse(veh);

    double dx = haut.x - p[9], dz = haut.z - p[11];
    double d = rac(dx * dx + dz * dz);
    double h = p[10] - haut.y;
    double rx = vh.x - vb.x, ry = vh.y - vb.y, rz = vh.z - vb.z;
    double vrel = rac(rx * rx + ry * ry + rz * rz);

    /* platform free, at the top, and no aircraft taxiing onto it (runway state 7) */
    int pret = I32(mgr, 0x7c) == -1 && F32(carrier, 0x1518) > -0.01f && I32(mgr, 0x24) != 7;
    if (engage && (I32(mgr, 0x7c) != -1 || d > 8.0))
        engage = 0;
    if (!engage && pret && d < 3.0 && vrel < 2.0 && h < 25.0)
        engage = 1;

    if (engage && h < 25.0)
        lacher_cargaison(carrier, veh, vcont, map, scene);
    int contact = roues_au_contact(veh);
    if (engage && d < 4.0 && vrel < 4.0 && ((contact == 1 && h < 8.0) || (contact == -1 && h < 3.0))) {
        if (poser(mgr, carrier, veh, map, peers, scene))
            return;
    }
    if (ticks > TICKS_MAX) {
        renvoyer_vers_piste(mgr, veh);
        return;
    }

    /* target: above the platform (higher when coming from far), or on it if engaged */
    double hc = engage ? -2.0 : (d < 40.0 ? H_ATTENTE : borne(H_ATTENTE + 0.25 * (d - 40.0), H_ATTENTE, 60.0));
    vec3d cible = vers_monde(c, (vec3d){ nm[9], nm[10] + hc, nm[11] });

    /* desired velocity = ship velocity + approach towards the platform; heading = direction of travel when far,
       then the platform's heading (ship stopped) or the ship's heading (ship under way) near the goal */
    double vx = vb.x, vz = vb.z;
    double ux = d > 0.01 ? dx / d : 0.0, uz = d > 0.01 ? dz / d : 0.0;
    double s = V_APPROCHE < 0.3 * d ? V_APPROCHE : 0.3 * d;
    vx += ux * s;
    vz += uz * s;
    double nv = rac(vx * vx + vz * vz);
    double lx = nv > 1.0 ? vx / nv : ux, lz = nv > 1.0 ? vz / nv : uz;
    double vb_h = rac((double)vb.x * vb.x + (double)vb.z * vb.z);
    double px = vb_h > 1.5 ? cap_bateau.x : cap_ascenseur.x, pz = vb_h > 1.5 ? cap_bateau.z : cap_ascenseur.z;
    double w = borne((d - 2.0) / 6.0, 0.0, 1.0);
    double ax = w * lx + (1.0 - w) * px, az = w * lz + (1.0 - w) * pz;
    double na = rac(ax * ax + az * az);
    if (na < 1e-6) { ax = px; az = pz; na = rac(ax * ax + az * az); }
    if (na < 1e-6) { ax = 0.0; az = 1.0; na = 1.0; }
    ax /= na;
    az /= na;

    I32(veh, 0x220) = 1;
    I32(veh, 0x2a0) = 3;
    F64(veh, 0x2a8) = cible.x;
    F64(veh, 0x2b0) = cible.y;
    F64(veh, 0x2b8) = cible.z;
    F32(veh, 0x2c0) = (float)ax;
    F32(veh, 0x2c4) = (float)az;
    F32(veh, 0x2c8) = (float)(vx * ax + vz * az);
    F32(veh, 0x2cc) = 1.0f;
    reglages_vol(veh, 0.6f, 1.0f);
    if (d < 60.0)
        train(veh, 1);

    U32(mgr, 0x9c) = (TAG_ETAT << 24) | (engage ? ENGAGE : 0) | (ticks & 0x7fffffu);
}

static void choisir(void *mgr, void *carrier, void *vcont, void *peers)
{
    u32 n = U32(mgr, 0x58);
    if (!n || !joueurs_a_bord(carrier, peers))
        return;
    void *def = PTR(carrier, 0x98);
    if (g_bay_index_attach(carrier, 1, 0xff) >= U32(def, 0x140))
        return;
    u32 refus = U32(mgr, 0x114);
    u32 banni = (refus >> 24) == TAG_REFUS ? (refus & 0xffffffu) : 0xffffffffu;
    double *c = corps0(carrier);
    double nm[12];
    matrice_ascenseur(carrier, mgr, nm);
    vec3d haut = vers_monde(c, (vec3d){ nm[9], nm[10], nm[11] });
    u32 cap = U32(mgr, 0x50), tete = U32(mgr, 0x54);
    u32 *file = (u32 *)PTR(mgr, 0x48);
    for (u32 i = 0; i < n; i++) {
        u32 *el = file + 3 * anneau(tete, i, cap);
        u32 id = el[0];
        if ((id & 0xffffffu) == banni)
            continue;
        void *v = vehicule(vcont, id);
        if (!v || !((fn_bool)VFN(v, 0xa0))(v))
            continue;
        double *p = corps0(v);
        double dx = haut.x - p[9], dz = haut.z - p[11];
        if (dx * dx + dz * dz > PORTEE * PORTEE)
            continue;
        g_queue_erase(AT(mgr, 0x48), i);
        u32 reste = U32(mgr, 0x58);
        cap = U32(mgr, 0x50);
        tete = U32(mgr, 0x54);
        file = (u32 *)PTR(mgr, 0x48);
        for (u32 k = 0; k < reste; k++)
            file[3 * anneau(tete, k, cap) + 2] = 0;
        I32(mgr, 0x84) = (int)id;
        U32(mgr, 0x9c) = TAG_ETAT << 24;
        train(v, 1);
        I32(v, 0x220) = 1;
        return;
    }
}

HIDDEN void hl_tick(void *mgr, void *carrier, void *vcont, void *map, void *peers, void *scene)
{
#ifdef TEMOIN
    (void)mgr; (void)carrier; (void)vcont; (void)map; (void)peers; (void)scene;
#else
    int id = I32(mgr, 0x84);
    if (id != -1) {
        void *v = vehicule(vcont, (u32)id);
        if (!v || I32(v, 0x220) != 1 || !((fn_bool)VFN(v, 0xa0))(v)) {
            liberer(mgr);
            return;
        }
        piloter(mgr, carrier, v, vcont, map, peers, scene);
        return;
    }
    choisir(mgr, carrier, vcont, peers);
#endif
}

/* hooks: game registers at the entry of the "returns" mode (0x1001acd94):
   r14 manager, r15 ship, r8 vehicle container, r13 map, rbx players, -0x80(%rbp) physics scene;
   rsp 16-byte aligned. Every register the game may still read is saved. */
__asm__(
    ".text\n"
    ".p2align 4\n"
    ".globl _hl_h1\n"
    "_hl_h1:\n"
    "    movq %r13, -0x70(%rbp)\n"
    "    pushq %rax\n pushq %rcx\n pushq %rdx\n pushq %rsi\n pushq %rdi\n"

##### FILE: workshop/fichiers/horloge/library_custom_1.lua #####
-- ============================================================================
--  GAME CLOCK  --  displays the game time at the bottom left of the vehicle
--  control screens (screen_veh_l / screen_veh_m / screen_veh_r).
--  No other screen is touched.
--
--  This is the game's internal clock (30 ticks per second of game time): if the game
--  lags, the clock slows down with it, so lag does not distort a measurement.
--  Read the clock at the start and at the end; the difference = duration in game time.
-- ============================================================================

function horloge_format(ticks)
	local s = ticks / 30
	local h = math.floor(s / 3600)
	local m = math.floor((s - 3600 * h) / 60)
	local r = s - 3600 * h - 60 * m
	if h > 0 then
		return string.format("%d:%02d:%04.1f", h, m, r)
	end
	return string.format("%d:%04.1f", m, r)
end

function horloge_dessiner(screen_w, screen_h)
	local texte = "GAME TIME  " .. horloge_format(update_get_logic_tick())
	local tw, th = update_ui_get_text_size(texte, 1000, 0)
	local w, h = math.min(tw + 8, screen_w - 8), th + 4
	local x, y = 4, screen_h - 4 - h
	update_ui_rectangle(x, y, w, h, color8(0, 0, 0, 210))
	update_ui_text(x + 4, y + 2, texte, w - 8, 0, color8(255, 255, 255, 255), 0)
end

-- transparent wrapper: same parameters, same return values; an error in the
-- clock cannot break the screen (pcall)
function horloge_update(...)
	local retours = table.pack(g_horloge_update_origine(...))
	local screen_w, screen_h = ...
	pcall(horloge_dessiner, screen_w, screen_h)
	return table.unpack(retours, 1, retours.n)
end

function horloge_begin_load(...)
	local retours = table.pack(g_horloge_begin_load_origine(...))

	local nom = nil
	if type(begin_get_screen_name) == "function" then
		local ok, valeur = pcall(begin_get_screen_name)
		if ok then nom = valeur end
	end

	if (nom == "screen_veh_l" or nom == "screen_veh_m" or nom == "screen_veh_r")
		and type(update) == "function"
		and g_horloge_update_origine == nil then
		g_horloge_update_origine = update
		update = horloge_update
	end

	return table.unpack(retours, 1, retours.n)
end

	g_horloge_begin_load_origine = begin_load
	begin_load = horloge_begin_load
end


##### FILE: workshop/fichiers/journal/library_custom_4.lua #####
-- ============================================================================
--  LANDING LOG  --  automatically times the returns of planes
--  and helicopters to this carrier, and displays the result at the bottom left
--  of the vehicle control screens (screen_veh_l / _m / _r), above the
--  game clock. Changes nothing in the game: it only reads states.
--
--  For each aircraft: the time it enters the queue, the time it
--  is given the runway, the time it touches the deck and the time it is stowed in the hangar (game
--  time, 30 ticks per second: lag does not distort anything). A "return"
--  starts when an aircraft enters the queue while nothing is in progress,
--  and ends when all of them are stowed. The result stays displayed until the
--  next return.
-- ============================================================================

g_journal = { appareils = {}, ordre = {}, debut = nil, fini = false, dernier_tick = nil, trou = 0 }  -- appareils = aircraft, ordre = order of arrival, debut = start, fini = finished, trou = longest gap between two frames

local FILE, PISTE, PONT, RANGE, VOL = 5, 1, 6, 4, 0  -- get_dock_state() values: FILE = queue, PISTE = runway, PONT = deck, RANGE = stowed in the hangar, VOL = flying

function journal_duree(ticks)
	if ticks == nil then return "--:--" end
	local s = math.floor(ticks / 30 + 0.5)
	return string.format("%d:%02d", math.floor(s / 60), s % 60)
end

function journal_genre(vehicule)
	local d = vehicule:get_definition_index()
	if d == e_game_object_type.chassis_air_wing_light or d == e_game_object_type.chassis_air_wing_heavy then
		return "avion"
	elseif d == e_game_object_type.chassis_air_rotor_light or d == e_game_object_type.chassis_air_rotor_heavy then
		return "helico"
	end
	return nil
end

function journal_nouveau(id, genre, t, etat)
	local j = g_journal
	if j.debut == nil or j.fini then
		j.appareils, j.ordre, j.debut, j.fini, j.trou = {}, {}, t, false, 0
	end
	local a = { id = id, genre = genre, file = t, etat = etat, ratees = 0 }
	if etat == PISTE then a.piste = t end
	if etat == PONT then a.pont = t end
	j.appareils[id] = a
	table.insert(j.ordre, a)
end

function journal_suivre(a, etat, t)
	if etat == a.etat then return end
	if etat == PISTE then
		a.piste = t
	elseif etat == PONT then
		a.pont = t
	elseif etat == RANGE then
		a.range = t
	elseif etat == FILE and a.etat == PISTE then
		a.ratees = a.ratees + 1                     -- missed approach: back to the queue
	elseif etat == VOL then
		a.abandon = true                            -- recall cancelled, or no room left
	end
	a.etat = etat
end

function journal_observer()
	local j = g_journal
	local t = update_get_logic_tick()
	if j.dernier_tick ~= nil and j.debut ~= nil and not j.fini and t - j.dernier_tick > j.trou then
		j.trou = t - j.dernier_tick
	end
	j.dernier_tick = t

	local bateau = update_get_screen_vehicle()
	if not bateau:get() then return end
	local id_bateau = bateau:get_id()

	-- new aircraft returning to this ship
	for i = 0, update_get_map_vehicle_count() - 1 do
		local v = update_get_map_vehicle_by_index(i)
		if v:get() then
			local etat = v:get_dock_state()
			if (etat == FILE or etat == PISTE or etat == PONT) and v:get_dock_parent_id() == id_bateau then
				local id = v:get_id()
				local a = (not j.fini) and j.appareils[id] or nil
				if a == nil then
					local genre = journal_genre(v)
					if genre then journal_nouveau(id, genre, t, etat) end
				elseif a.abandon and a.range == nil then
					a.abandon, a.etat = false, etat         -- recalled again during the same return
				end
			end
		end
	end

	-- aircraft already being tracked
	if j.debut ~= nil and not j.fini then
		local reste = 0
		for _, a in ipairs(j.ordre) do
			if a.range == nil and not a.abandon then
				local v = update_get_map_vehicle_by_id(a.id)
				if v:get() then
					journal_suivre(a, v:get_dock_state(), t)
				else
					a.abandon = true                    -- aircraft destroyed
				end
				if a.range == nil and not a.abandon then reste = reste + 1 end
			end
		end
		if reste == 0 and #j.ordre > 0 then j.fini = true end
	end
end

function journal_resume()
	local j = g_journal
	local lignes = {}
	local genres = { avion = 0, helico = 0 }  -- keys: avion = plane, helico = helicopter
	local pistes, approches, ponts, ratees, fin = {}, {}, {}, 0, nil
	for k, a in ipairs(j.ordre) do
		genres[a.genre] = genres[a.genre] + 1
		local nom = (a.genre == "avion") and "plane " or "copter"
		local etat = a.abandon and " abandon" or ""
		-- short (< 36 characters): anything longer wraps onto a new line on the game screen (observed in game)
		table.insert(lignes, string.format("%d %s  %5s  %5s  %5s%s", k, nom,
			journal_duree(a.piste and a.piste - j.debut), journal_duree(a.pont and a.pont - j.debut),
			journal_duree(a.range and a.range - j.debut), etat))
		if a.piste then table.insert(pistes, a.piste) end
		if a.piste and a.pont then table.insert(approches, a.pont - a.piste) end
		if a.pont and a.range then table.insert(ponts, a.range - a.pont) end
		if a.range and (fin == nil or a.range > fin) then fin = a.range end
		ratees = ratees + a.ratees
	end
	local function moyenne(liste)
		if #liste == 0 then return nil end
		local s = 0
		for _, x in ipairs(liste) do s = s + x end
		return s / #liste
	end
	table.sort(pistes)
	local ecarts = {}
	for k = 2, #pistes do table.insert(ecarts, pistes[k] - pistes[k - 1]) end

	local function compte(n, mot) return string.format("%d %s%s", n, mot, (n > 1) and "s" or "") end
	local quoi = compte(genres.avion, "plane")
	if genres.helico > 0 then quoi = quoi .. " + " .. compte(genres.helico, "copter") end
	if j.fini then
		table.insert(lignes, string.format("DONE in %s : %s", journal_duree(fin and fin - j.debut), quoi))
	else
		table.insert(lignes, string.format("ONGOING %s : %s", journal_duree(j.dernier_tick - j.debut), quoi))
	end
	table.insert(lignes, string.format("runway every %s  missed %d", journal_duree(moyenne(ecarts)), ratees))
	table.insert(lignes, string.format("approach %s  deck>hangar %s", journal_duree(moyenne(approches)),
		journal_duree(moyenne(ponts))))
	if j.trou > 150 then
		table.insert(lignes, string.format("! screen frozen %s : unreliable", journal_duree(j.trou)))
	end
	return lignes
end

function journal_dessiner(screen_w, screen_h)
	if g_journal.debut == nil then return end
	local lignes = journal_resume()
	table.insert(lignes, 1, "         runway   deck hangar")
	table.insert(lignes, 1, "LANDING LOG (game time)")
	local _, hh = update_ui_get_text_size("GAME TIME", 1000, 0)
	local w = 0
	local th = hh
	for _, l in ipairs(lignes) do
		local tw, h = update_ui_get_text_size(l, 1000, 0)
		if tw > w then w = tw end
		th = h
	end
	w = math.min(w + 8, screen_w - 8)
	local h = #lignes * (th + 1) + 4
	local x, y = 4, screen_h - 4 - (hh + 4) - 2 - h          -- just above the clock
	update_ui_rectangle(x, y, w, h, color8(0, 0, 0, 210))
	for k, l in ipairs(lignes) do
		local couleur = (k == 1) and color8(255, 200, 80, 255) or color8(255, 255, 255, 255)
		update_ui_text(x + 4, y + 2 + (k - 1) * (th + 1), l, w - 8, 0, couleur, 0)
	end
end

-- transparent wrapper: same parameters, same return values; an error in the
-- log cannot break the screen (pcall)
function journal_update(...)
	local retours = table.pack(g_journal_update_origine(...))
	local screen_w, screen_h = ...

-- ============================================================================
--  PERFORMANCE LOG  --  records, during the game and into a file, what is needed to
--  plot the lag curves afterwards (the plotting tool, not included).
--  Displays nothing and changes nothing in the game: it only reads states.
--
--  Each screen (and the piloting display) writes with print():
--    - every 10 s, an "E" line (heartbeat): real time, game tick,
--      screen name, number of calls to update since the previous heartbeat,
--      whether the screen is on, whether the player is on this screen;
--    - for a few screens only (DONNEES), every 5 s, a "D" line
--      written as soon as the real second changes (so the game tick is accurate):
--      units per team and per type, missiles and shells in flight, wrecks.
--  The original game discards what print() writes; the "ecriture_releve" module
--  (a patch of the game executable) sends it to
--  <HOME>/Library/Application Support/Carrier Command 2/releve_cc2.txt.
--  The print() calls of mods (a replacement mod makes many) also go through, but at most
--  20 per minute and per screen; the rest is only counted (x= in the "E" line).
-- ============================================================================

local PREFIXE = "CC2R1"
local DONNEES = { holomap_screen = true, screen_veh_m = true, screen_nav_m = true, hud = true }  -- screens that also write the "D" lines
local CATEGORIE = {                 -- e_game_object_type -> column of q<team>=
	[0] = 1,                        -- aircraft carrier
	[2] = 2, [4] = 2, [6] = 2, [58] = 2,     -- land units (wheeled, robot dog)
	[8] = 3, [10] = 3,              -- planes
	[12] = 4, [14] = 4,             -- helicopters
	[16] = 5, [57] = 5, [77] = 5, [79] = 5,  -- barges, boats, ships
	[59] = 6,                       -- island turrets
}
local AUTRE, HANGAR, CACHES, COLONNES = 7, 8, 9, 9  -- AUTRE = other, HANGAR = in hangar, CACHES = hidden, COLONNES = number of columns
local DOCK_HANGAR = 4
local MAX_PRINT_MODS = 20

local function lire(f, ...)
	if type(f) ~= "function" then return nil end
	local ok, v = pcall(f, ...)
	if ok then return v end
	return nil
end

local function num(v)
	if v == true then return 1 elseif v == false then return 0 end
	if type(v) == "number" then return math.floor(v) end
	return -1
end

function releve_ecrire(ligne)
	g_releve_print_origine(PREFIXE .. " " .. ligne .. " FIN")
end

-- print() of mods: lets it through, but limited (the file must not blow up)
function releve_print(...)
	local r = g_releve
	if r ~= nil and r.e ~= nil then
		local minute = r.e // 60
		if minute ~= r.minute_mods then r.minute_mods, r.print_mods = minute, 0 end
		if r.print_mods >= MAX_PRINT_MODS then
			r.print_mods_jetes = r.print_mods_jetes + 1
			return
		end
		r.print_mods = r.print_mods + 1
	end
	return g_releve_print_origine(...)
end

function releve_donnees(e, t)
	local r = g_releve
	local equipes = {}
	local n = lire(update_get_map_vehicle_count) or 0
	for i = 0, n - 1 do
		local v = update_get_map_vehicle_by_index(i)
		if v ~= nil and v:get() then
			local equipe = num(lire(v.get_team, v))
			local q = equipes[equipe]
			if q == nil then
				q = {}
				for k = 1, COLONNES do q[k] = 0 end
				equipes[equipe] = q
			end
			local col = CATEGORIE[lire(v.get_definition_index, v)] or AUTRE
			if col ~= AUTRE and lire(v.get_dock_state, v) == DOCK_HANGAR then col = HANGAR end
			q[col] = q[col] + 1
			if lire(v.get_is_visible, v) == false then q[CACHES] = q[CACHES] + 1 end
		end
	end
	local morceaux = { string.format("D e=%d t=%d s=%s moi=%d nv=%d mis=%d pro=%d epa=%d", e, t, r.nom,
		num(lire(update_get_screen_team_id)), n, num(lire(update_get_missile_count)),
		num(lire(update_get_projectile_count)), num(lire(update_get_map_destroyed_vehicle_count))) }
	if r.vehicule ~= nil then
		local pil = lire(r.vehicule.get_definition_index, r.vehicule)
		if pil ~= nil then table.insert(morceaux, "pil=" .. num(pil)) end
	end
	local ids = {}
	for equipe in pairs(equipes) do table.insert(ids, equipe) end
	table.sort(ids)
	for _, equipe in ipairs(ids) do
		table.insert(morceaux, string.format("q%d=%s", equipe, table.concat(equipes[equipe], ",")))
	end
	releve_ecrire(table.concat(morceaux, " "))
end

function releve_battement(e, t)
	local r = g_releve
	releve_ecrire(string.format("E e=%d t=%d s=%s n=%d a=%d f=%d x=%d", e, t, r.nom, r.appels,
		num(lire(update_get_screen_state_active)), num(lire(update_get_is_focus_local)), r.print_mods_jetes))
	r.appels, r.print_mods_jetes = 0, 0
end

function releve_observer()
	local r = g_releve
	r.appels = r.appels + 1
	if not r.donnees and r.appels % 8 ~= 1 then return end      -- the other screens check the time 1 call out of 8
	local e = update_get_time_since_epoch()
	if type(e) ~= "number" then return end
	e = math.floor(e)
	if e == r.e then return end
	local premiere = (r.e == nil)
	r.e = e
	local t = update_get_logic_tick()
	if premiere then
		if r.donnees then releve_ecrire(string.format("S e=%d t=%d s=%s", e, t, r.nom)) end
		r.prochain = e
	end
	if r.donnees and e % 5 == 0 then releve_donnees(e, t) end
	if e >= r.prochain then
		releve_battement(e, t)
		r.prochain = e + 10 - ((e - r.decalage) % 10)
	end
end

local function releve_passe(...)
	pcall(releve_observer)
	return ...
end

-- transparent wrapper: same parameters, same return values; an error in the
-- performance log cannot break the screen (pcall)
function releve_update(...)
	if g_releve.nom == "hud" then g_releve.vehicule = select(6, ...) end
	return releve_passe(g_releve_update_origine(...))
end

function releve_begin_load(...)
	local retours = table.pack(g_releve_begin_load_origine(...))

	local nom = lire(begin_get_screen_name)
	if type(nom) ~= "string" then
		nom = "autre"  -- "autre" = other; written to the file as s=autre (file-format key, kept as is)
		if type(update) == "function" and debug ~= nil and debug.getinfo ~= nil then
			local ok, info = pcall(debug.getinfo, update, "u")
			if ok and info and info.nparams and info.nparams >= 7 then nom = "hud" end
		end
	end

	if type(update) == "function" and g_releve_update_origine == nil
		and type(update_get_time_since_epoch) == "function" and type(update_get_logic_tick) == "function"
		and type(g_releve_print_origine) == "function" then
		local somme = 0
		for k = 1, #nom do somme = somme + nom:byte(k) end
		g_releve = { nom = nom, donnees = DONNEES[nom] == true, appels = 0, decalage = somme % 10,
			print_mods = 0, print_mods_jetes = 0 }
		g_releve_update_origine = update
		update = releve_update
	end

	return table.unpack(retours, 1, retours.n)
end

if type(print) == "function" and g_releve_print_origine == nil then
	g_releve_print_origine = print
	print = releve_print
end

if type(begin_load) == "function" and g_releve_begin_load_origine == nil then
	g_releve_begin_load_origine = begin_load
	begin_load = releve_begin_load
end

-- ============================================================================
--  SQUADRON SPEED  --  ECO / NORMAL / FLANK bar at the bottom right of the
--  vehicle control screens. A click chooses the cruise speed of
--  all the AI-piloted planes of the team:
--    ECO    = original game behaviour (about 50 % power)

--

def compiler(temoin, dossier):

def lire_objet(o):
    magic, cpu, _, typ, ncmds, _, _, _ = struct.unpack_from("<IiiIIIII", o, 0)
    if magic != 0xfeedfacf or cpu != 0x01000007 or typ != 1:
        raise Erreur("x86_64 Mach-O object expected")
    pos, sections, syms = 32, [], []
    for _ in range(ncmds):
        cmd, taille = struct.unpack_from("<II", o, pos)
        if cmd == 0x19:
            nsect = struct.unpack_from("<I", o, pos + 64)[0]
            for k in range(nsect):
                s = pos + 72 + 80 * k
                nom = o[s:s + 16].rstrip(b"\0").decode()
                seg = o[s + 16:s + 32].rstrip(b"\0").decode()
                addr, size, off, align, reloff, nreloc = struct.unpack_from("<QQIIII", o, s + 32)
                sections.append(dict(nom=nom, seg=seg, addr=addr, size=size, off=off, align=align,
                                     reloff=reloff, nreloc=nreloc))
        elif cmd == 0x2:
            symoff, nsyms, stroff, _ = struct.unpack_from("<IIII", o, pos + 8)
            for k in range(nsyms):
                strx, ntype, nsect, _, val = struct.unpack_from("<IBBHQ", o, symoff + 16 * k)
                nom = o[stroff + strx:o.index(b"\0", stroff + strx)].decode()
                syms.append(dict(nom=nom, type=ntype, sect=nsect, val=val))
        pos += taille
    return sections, syms


def relier(o, base):
    sections, syms = lire_objet(o)
    for s in sections:
        if s["nom"] not in SECTIONS_PRISES and s["seg"] != "__LD":
            if s["size"]:
                raise Erreur("unexpected section %s,%s" % (s["seg"], s["nom"]))
    prises = [s for s in sections if s["nom"] in SECTIONS_PRISES and s["size"]]
    adresse = base
    for s in prises:
        a = 1 << s["align"]
        adresse = (adresse + a - 1) // a * a
        s["neuve"] = adresse
        adresse += s["size"]
    fin = adresse
    image = bytearray(b"\xcc" * (fin - base))     # int3 between the sections
    for s in prises:
        image[s["neuve"] - base:s["neuve"] - base + s["size"]] = o[s["off"]:s["off"] + s["size"]]

    def sect_de(n):      # section number 1.. of the object
        return sections[n - 1]

    def adresse_symbole(sym):
        if sym["type"] & 0x0e == 0x0e:        # N_SECT: defined in the graft
            s = sect_de(sym["sect"])
            if "neuve" not in s:
                raise Erreur("symbol %s in a section that was not taken" % sym["nom"])
            return s["neuve"] + sym["val"] - s["addr"]
        if sym["nom"] not in JEU:
            raise Erreur("unknown symbol: " + sym["nom"])
        return JEU[sym["nom"]]

    relocs = []
    for s in prises:
        for k in range(s["nreloc"]):
            r_addr, info = struct.unpack_from("<iI", o, s["reloff"] + 8 * k)
            num, pcrel, longueur, externe, typ = info & 0xffffff, (info >> 24) & 1, (info >> 25) & 3, (info >> 27) & 1, info >> 28
            if not pcrel or longueur != 2 or typ not in (1, 2, 6, 7, 8):
                raise Erreur("unsupported relocation: type %d pcrel %d length %d at %#x" % (typ, pcrel, longueur, r_addr))
            biais = {1: 0, 2: 0, 6: 1, 7: 2, 8: 4}[typ]
            p_neuf = s["neuve"] + r_addr
            p_vieux = s["addr"] + r_addr
            stocke = struct.unpack_from("<i", image, p_neuf - base)[0]
            if externe:
                # the assembler stores (addend - bias): real target = symbol + stored + bias (like ld64)
                sym = syms[num]
                cible = adresse_symbole(sym) + stocke + biais
                nom = sym["nom"]
            else:
                cs = sect_de(num)
                cible_vieille = p_vieux + 4 + biais + stocke
                if "neuve" not in cs or not cs["addr"] <= cible_vieille <= cs["addr"] + cs["size"]:
                    raise Erreur("local relocation outside section at %#x" % p_vieux)
                cible = cs["neuve"] + cible_vieille - cs["addr"]
                nom = cs["nom"]
            disp = cible - (p_neuf + 4 + biais)
            if not -2 ** 31 <= disp < 2 ** 31:
                raise Erreur("displacement too large at %#x" % p_neuf)
            struct.pack_into("<i", image, p_neuf - base, disp)
            relocs.append((p_neuf, cible, nom))
    points = {}
    for sym in syms:
        if sym["type"] & 0x0e == 0x0e and sym["type"] & 0x01:
            points[sym["nom"]] = adresse_symbole(sym)
    return bytes(image), points, relocs


def ecrire_module(chemin, nom, titre, doc, retouches, md5_modifie):
    lignes = ['# -*- coding: utf-8 -*-', '"""' + doc.strip() + '\n"""',
              'NOM = %r' % nom, 'TITRE = %r' % titre, 'TYPE = "programme"',
              '# md5 of the executable with this module alone: %s' % md5_modifie, '', 'RETOUCHES = [']
    for va, avant, apres, commentaire in retouches:
        lignes.append('    # %s' % commentaire)
        lignes.append('    ("imm", %#x, "%s", "%s"),' % (va, avant, apres))
    lignes.append(']')
    open(chemin, "w").write("\n".join(lignes) + "\n")


def generer(temoin, programme, dossier):
    o = compiler(temoin, dossier)
    image, points, relocs = relier(o, ZONE_DEBUT)
    if ZONE_DEBUT + len(image) > ZONE_FIN:
        raise Erreur("the graft exceeds the free zone")
    off = ZONE_DEBUT - BASE_FICHIER
    retouches = [(ZONE_DEBUT, programme[off:off + len(image)].hex(), image.hex(),
                  "added code (%d bytes) in place of ImGui::ShowDemoWindow, never called" % len(image))]
    for va, attendu, point, commentaire in BRANCHEMENTS:
        o2 = va - BASE_FICHIER
        n = len(attendu) // 2
        if programme[o2:o2 + n].hex() != attendu:
            raise Erreur("unexpected original bytes at %#x" % va)
        saut = b"\xe9" + struct.pack("<i", points[point] - (va + 5))
        retouches.append((va, attendu, (saut + b"\x90" * (n - 5)).hex(),
                          "%s -> jmp %s (%#x)" % (commentaire, point, points[point])))
    modifie = bytearray(programme)
    for va, avant, apres, _ in retouches:
        modifie[va - BASE_FICHIER:va - BASE_FICHIER + len(avant) // 2] = bytes.fromhex(apres)
    return retouches, hashlib.md5(modifie).hexdigest(), points, relocs, len(image)

    (0x1001af85b, "jbe", 0x1001afb05, BLOC_PISTE, "state 2: test the runway during the rest of the rear turn"),

RIP = [
    # (address, mnemonic, old constant, new constant, expected values of the new one)

    (0x1001afae7, "addps", 0x10286b900, 0x10286b930, (-1000.0, -1800.0), "state 5: same (back to the original frame)"),

"""Generates the "pont_retour_x2" module of the workshop from the disassembly (never by hand).

After touchdown, a plane taxis on the deck in 3 legs (runway states 5, 6, 7 of
c_vehicle_server_carrier::docking_manager_air::update): roll back from the front of the runway to taxi_a
(~125 m), taxi_a -> lift_upper_wait, then -> elevator platform. On each tick, the progress along the
leg increases by v / (L x 30). The three "mulss [30.0]" of these states are distinct instructions:
we make them read 15.0 (constant already present) -> taxiing goes 2x faster, for these three
legs only. The runway stays reserved until the elevator, as in the original game.

The generator checks, on the original executable (sauvegarde/build_mac_origine):
  - each targeted instruction: address, mnemonic, constant read (30.0);
  - the new constant (15.0), read-only;
  - that each instruction is reached ONLY from its own state: complete list of the function's jumps
    that lead to it, and previous instruction (whether it falls through into it or not).

TRENTE, QUINZE = 0x102869898, 0x1028699cc

    (0x1001ae07c, {0x1001ae05e, 0x1001ae06f}, True, "state 6 : taxi_a -> lift_upper_wait (~17 m)"),

]

#!/usr/bin/env python3
"""Simulation of the CC2 aircraft landing queue (logic read in docking_manager_air::update).

Model: waiting planes follow the pattern at 80 m/s (abscissa s along the lap, 0 = the "gate" at the
bottom of the rear turn). A plane is given the runway if the runway is free AND it is inside the
assignment window. The runway stays occupied O seconds (approach + touchdown + taxiing to the elevator).
Original game: window = the instant the plane passes the gate (one tick per lap).
"""
import random

V = 80.0              # m/s, holding speed (override_target_velocity = 80)
TICK = 1 / 30.0
QUART = 1200 * 3.14159265 / 2          # quarter of a turn, radius 1200 m
AVANT = 1031.0        # part of state 1 within 1000 m of the gate
APRES = 1684.0        # part of state 2 where the distance test passes (z < -2000)

CIRCUITS = {
    "origine": 2 * 2400 + 2 * 1200 * 3.14159265,    # oval: 2 straight lines of 2400 m + 2 half-circles
    "cercle": 2 * 1200 * 3.14159265,                # both turns on the same circle
}


def simuler(tour, fenetre, O, departs, horizon=3000):
    """departs: starting abscissae of the planes (m, counted BEFORE the gate). Returns the instants
    of runway assignment and the final release time."""
    s = [(-d) % tour for d in departs]          # position along the lap
    attente = list(range(len(s)))
    libre_a = 0.0
    attrib = []
    t = 0.0
    while attente and t < horizon:
        if t >= libre_a:
            for i in attente:
                a, b = s[i], (s[i] + V * TICK)
                if fenetre:
                    dedans = s[i] >= tour - AVANT or s[i] <= APRES
                else:
                    dedans = a < tour <= b or (b >= tour)            # passing the gate during this tick
                if dedans:
                    attrib.append(t)
                    attente.remove(i)
                    libre_a = t + O
                    break
        for i in attente:
            s[i] = (s[i] + V * TICK) % tour
        t += TICK
    return attrib, (attrib[-1] + O if len(attrib) == len(departs) else None)


def scenario(nom, departs_fn, essais):
    print("\n" + nom)
    print("  landing duration O :        90 s    110 s    130 s    150 s    170 s")
    for titre, tour, fen in (("original game          ", CIRCUITS["origine"], False),
                             ("A widened window       ", CIRCUITS["origine"], True),
                             ("B A + circular pattern ", CIRCUITS["cercle"], True)):
        ligne = []
        for O in (90, 110, 130, 150, 170):
            tot = []
            for k in range(essais):
                random.seed(k)
                _, fin = simuler(tour, fen, O, departs_fn(tour))
                tot.append(fin)
            ligne.append(sum(tot) / len(tot))
        print("  %s  %s" % (titre, "  ".join("%5.0f s" % x for x in ligne)))


if __name__ == "__main__":
    print("Lap of the pattern: original %.0f m (%.0f s), circle %.0f m (%.0f s)"
          % (CIRCUITS["origine"], CIRCUITS["origine"] / V, CIRCUITS["cercle"], CIRCUITS["cercle"] / V))
    # 4 planes coming back together, in a line 300 m apart, the first one 2 km from the gate
    scenario("4 grouped planes (time between arrival near the gate and the 4th on the elevator)",
             lambda tour: [2000 + 300 * k for k in range(4)], 1)
    # 4 planes arriving at different times: random positions on the pattern
    scenario("4 planes scattered at random on the pattern (average of 400 trials)",
             lambda tour: [random.uniform(0, tour) for _ in range(4)], 400)

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Patch workshop for Carrier Command 2 (Mac build): one tool for all the changes.

Each change is a MODULE (a file in outil/modules/) of one of these two types:
  - "programme" : byte patches in the game executable (MacOS/build_mac);
  - "fichier"   : replacement of a file in rom_0 (Lua script, graphics effect...).
Modules can be enabled and removed independently; the tool checks that they never touch
the same bytes or the same file.

Usage:
  patchs_cc2.py                       -> status of each module
  patchs_cc2.py --activer NAME [NAME..] -> enables these modules
  patchs_cc2.py --retirer NAME [NAME..] -> removes them
  patchs_cc2.py --origine             -> removes everything (original executable and files)
  patchs_cc2.py --menu                -> menu (this is what the double-click launcher runs)
For tests: --programme PATH, --rom FOLDER, --modules FOLDER, --sauvegarde PATH
(they work on copies; the "game closed" check is then skipped).

Safeguards:
  - refuses to run while the game is running;
  - refuses to touch the executable unless it is exactly the one that was analysed (once all
    modules are reverted to original, its md5 must be MD5_ORIGINE): after a Steam update, the
    analysis has to be redone;
  - a module whose bytes are neither all original nor all modified is "damaged": refused;
  - keeps a copy of the original executable (rebuilt, hence always the true one) and of the
    original files that were replaced; writes through a temporary file; reads back and checks.
"""
import hashlib, importlib.util, os, shutil, struct, subprocess, sys

ICI = os.path.dirname(os.path.abspath(__file__))
DOSSIER = os.path.dirname(ICI)
APP = os.path.expanduser("~/Library/Application Support/Steam/steamapps/common/Carrier Command 2/carriercommand.app/Contents")
PROGRAMME = os.path.join(APP, "MacOS", "build_mac")
ROM = os.path.join(APP, "Resources", "rom_0")
MODULES = os.path.join(ICI, "modules")
FICHIERS = os.path.join(DOSSIER, "fichiers")
SAUVEGARDE = os.path.join(DOSSIER, "sauvegarde", "build_mac_origine")
SAUVEGARDE_ROM = os.path.join(DOSSIER, "sauvegarde", "rom_0")

MD5_ORIGINE = "b0107170ddaf0245cbfee8a2d876aeb6"   # build_mac of 2026-01-05 (Steam)
BASE = 0x100000000                                   # x86_64-only binary: offset = address - BASE


class Refus(Exception):
    pass


# ---------------------------------------------------------------- common helpers

def md5(donnees):
    return hashlib.md5(donnees).hexdigest()


def sha256(donnees):
    return hashlib.sha256(donnees).hexdigest()


def lire(chemin):
    with open(chemin, "rb") as f:
        return f.read()


def ecrire(chemin, donnees):
    tmp = chemin + ".tmp_patchs_cc2"
    with open(tmp, "wb") as f:
        f.write(donnees)
    if os.path.exists(chemin):
        shutil.copymode(chemin, tmp)
    os.replace(tmp, chemin)


def lire_f32(donnees, va):
    return struct.unpack_from("<f", donnees, va - BASE)[0]


# ---------------------------------------------------------------- modules

def charger_modules(dossier):
    modules = []
    for nom in sorted(os.listdir(dossier)):
        if not nom.endswith(".py") or nom.startswith("_"):
            continue
        spec = importlib.util.spec_from_file_location("module_cc2_" + nom[:-3], os.path.join(dossier, nom))
        m = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(m)
        if m.TYPE not in ("programme", "fichier"):
            raise Refus("module %s: unknown type %r" % (nom, m.TYPE))
        modules.append(m)
    noms = [m.NOM for m in modules]
    if len(set(noms)) != len(noms):
        raise Refus("two modules have the same name")
    return modules


def retouches(module, donnees):
    """List of (offset, original bytes, modified bytes) of a "programme" module, checked.

    The module provides RETOUCHES: tuples
      ("imm", address, original_bytes_hex, modified_bytes_hex)
      ("rip", address, original_bytes_hex, old_constant, new_constant, factor)
    "rip" = instruction that reads a constant by RIP-relative address (disp32 at the end of the
    instruction): we redirect it to another constant already present in the game; if factor is
    not None, we check that new value == factor x old value (exact float).
    """
    liste = []
    for r in module.RETOUCHES:
        if r[0] == "imm":
            _, va, avant, apres = r
            avant, apres = bytes.fromhex(avant), bytes.fromhex(apres)
            if len(avant) != len(apres):
                raise Refus("%s: patch %#x changes the length" % (module.NOM, va))
        elif r[0] == "rip":
            _, va, avant, vieux, neuf, facteur = r
            avant = bytes.fromhex(avant)
            fin = va + len(avant)
            if int.from_bytes(avant[-4:], "little", signed=True) != vieux - fin:
                raise Refus("%s: the instruction at %#x does not read %#x" % (module.NOM, va, vieux))
            if facteur is not None and lire_f32(donnees, neuf) != facteur * lire_f32(donnees, vieux):
                raise Refus("%s: %#x is not %s x %#x" % (module.NOM, neuf, facteur, vieux))
            apres = avant[:-4] + (neuf - fin).to_bytes(4, "little", signed=True)
        else:
            raise Refus("%s: unknown patch type %r" % (module.NOM, r[0]))
        liste.append((va - BASE, avant, apres))
    return liste


def verifier_chevauchements(tables):
    occupe = {}
    for nom, table in tables.items():
        for o, a, _ in table:
            for i in range(o, o + len(a)):
                if i in occupe and occupe[i] != nom:
                    raise Refus("modules %s and %s touch the same bytes (%#x)"
                                % (occupe[i], nom, i + BASE))
                occupe[i] = nom


def etat_programme(donnees, table):
    if all(donnees[o:o + len(a)] == a for o, a, _ in table):
        return "retire"
    if all(donnees[o:o + len(b)] == b for o, _, b in table):
        return "actif"
    return "abime"


def etat_fichier(module, rom):
    cible = os.path.join(rom, module.CIBLE)
    if not os.path.isfile(cible):
        return "absent"
    h = sha256(lire(cible))
    if h == sha256(lire(os.path.join(FICHIERS, module.SOURCE))):
        return "actif"
    if h == module.ORIGINE_SHA256:
        return "retire"
    return "inconnu"


# ---------------------------------------------------------------- the work

class Atelier:
    def __init__(self, programme, rom, dossier_modules, sauvegarde, sauvegarde_rom):
        self.programme, self.rom = programme, rom
        self.sauvegarde, self.sauvegarde_rom = sauvegarde, sauvegarde_rom
        self.modules = {m.NOM: m for m in charger_modules(dossier_modules)}
        cibles = [m.CIBLE for m in self.modules.values() if m.TYPE == "fichier"]
        if len(set(cibles)) != len(cibles):
            raise Refus("two modules replace the same file")
        self.donnees = bytearray(lire(programme))
        self.tables = {n: retouches(m, self.donnees) for n, m in self.modules.items() if m.TYPE == "programme"}
        verifier_chevauchements(self.tables)

    def original_reconstruit(self):
        d = bytearray(self.donnees)
        for table in self.tables.values():
            for o, a, _ in table:
                d[o:o + len(a)] = a
        return d

    def programme_reconnu(self):
        return md5(self.original_reconstruit()) == MD5_ORIGINE

    def etats(self):
        res = {}
        for n, m in self.modules.items():
            res[n] = etat_programme(self.donnees, self.tables[n]) if m.TYPE == "programme" else etat_fichier(m, self.rom)
        return res

    def appliquer(self, voulus):
        """voulus: {name: True (active) / False (removed)}. Does nothing if everything is already in the wanted state."""
        etats = self.etats()
        for n in voulus:
            if n not in self.modules:
                raise Refus("unknown module: %s (known: %s)" % (n, ", ".join(self.modules)))
        a_faire = {n: v for n, v in voulus.items() if etats[n] != ("actif" if v else "retire")}
        if not a_faire:
            return []
        prog = [n for n in a_faire if self.modules[n].TYPE == "programme"]
        if prog:
            if not self.programme_reconnu():
                raise Refus("the game executable is not the one that was analysed (Steam update?). Nothing was touched.")
            for n in prog:
                if etats[n] == "abime":
                    raise Refus("module %s is damaged (neither all original nor all modified). Nothing was touched." % n)
        for n in a_faire:
            m = self.modules[n]
            if m.TYPE == "fichier" and etats[n] not in ("actif", "retire"):
                raise Refus("%s: file %s contains something other than the original or our version. Nothing was touched."
                            % (n, m.CIBLE))

        if prog:
            if not os.path.isfile(self.sauvegarde):
                os.makedirs(os.path.dirname(self.sauvegarde), exist_ok=True)
                ecrire(self.sauvegarde, bytes(self.original_reconstruit()))
            for n in prog:
                for o, a, b in self.tables[n]:
                    self.donnees[o:o + len(a)] = b if a_faire[n] else a
            ecrire(self.programme, bytes(self.donnees))
            self.donnees = bytearray(lire(self.programme))

        for n in a_faire:
            m = self.modules[n]
            if m.TYPE != "fichier":
                continue
            cible = os.path.join(self.rom, m.CIBLE)
            copie = os.path.join(self.sauvegarde_rom, m.CIBLE)
            if a_faire[n]:
                if not os.path.isfile(copie):
                    os.makedirs(os.path.dirname(copie), exist_ok=True)
                    shutil.copy2(cible, copie)
                ecrire(cible, lire(os.path.join(FICHIERS, m.SOURCE)))
            else:
                if os.path.isfile(copie) and sha256(lire(copie)) == m.ORIGINE_SHA256:
                    ecrire(cible, lire(copie))
                elif m.ORIGINE_SHA256 == sha256(b""):
                    ecrire(cible, b"")
                else:
                    raise Refus("%s: no copy of the original %s to restore." % (n, m.CIBLE))

        apres = self.etats()
        rates = [n for n, v in a_faire.items() if apres[n] != ("actif" if v else "retire")]
        if rates:
            raise Refus("read-back does not give the expected state for: " + ", ".join(rates))
        return sorted(a_faire)

LIBELLES = {"actif": "ACTIVE", "retire": "removed (original game)", "abime": "DAMAGED",

def afficher(atelier):

    for n, e in atelier.etats().items():
        m = atelier.modules[n]
        print("    %-14s %-24s %s" % (n, LIBELLES[e], m.TITRE))

AVEC_VALEUR = ("--programme", "--rom", "--modules", "--sauvegarde", "--sauvegarde-rom")


def lire_arguments(argv):
    """Strict parsing: any word that is not understood is refused (never ignored, so that we do
    not silently fall back on the real game when we meant to work on a copy)."""
    opts = {"activer": [], "retirer": [], "origine": False, "menu": False}
    i, liste = 0, None
    while i < len(argv):
        a = argv[i]
        if a in AVEC_VALEUR:
            if i + 1 >= len(argv) or argv[i + 1].startswith("--"):
                raise Refus("missing path after %s" % a)
            opts[a] = argv[i + 1]
            i, liste = i + 2, None
            continue
        if a in ("--activer", "--retirer"):
            liste = opts[a[2:]]
        elif a in ("--origine", "--menu"):
            opts[a[2:]], liste = True, None
        elif liste is not None and not a.startswith("-"):
            liste.append(a)
        else:
            raise Refus("unrecognised argument: %r" % a)
        i += 1
    return opts


def ouvrir(opts):
    programme, rom = opts.get("--programme", PROGRAMME), opts.get("--rom", ROM)
    if programme == PROGRAMME and rom == ROM and \
            subprocess.run(["pgrep", "-x", "build_mac"], capture_output=True).returncode == 0:
        raise Refus("close Carrier Command 2 first.")
    return Atelier(programme, rom, opts.get("--modules", MODULES),
                   opts.get("--sauvegarde", SAUVEGARDE), opts.get("--sauvegarde-rom", SAUVEGARDE_ROM))

    if faits:
        print("\n  OK: " + ", ".join(("%s %s" % (n, "enabled" if voulus[n] else "removed")) for n in faits)
              + ". Restart the game to see the effect.")
    else:
        print("\n  Nothing to do: already in that state.")


def menu(opts):
    """Menu for the double-click launcher: offers to enable or remove each module."""
    while True:
        atelier = ouvrir(opts)
        afficher(atelier)
        etats = atelier.etats()
        choix = []
        for n, e in etats.items():
            if e == "actif":
                choix.append(("Remove   " + atelier.modules[n].TITRE, {n: False}))
            elif e == "retire":
                choix.append(("Enable   " + atelier.modules[n].TITRE, {n: True}))
        choix.append(("Restore everything to original", {n: False for n in atelier.modules}))
        print("\n  What do you want to do?")
        for k, (texte, _) in enumerate(choix, 1):
            print("    %d. %s" % (k, texte))
        print("    q. Quit")
        try:
            rep = input("\n  Your choice: ").strip().lower()
        except EOFError:
            return 0
        if rep in ("q", ""):
            return 0
        if not rep.isdigit() or not 1 <= int(rep) <= len(choix):
            print("\n  Unknown choice.")
            continue
        faire(atelier, choix[int(rep) - 1][1])


def main(argv):
    print("\n  PATCH WORKSHOP - Carrier Command 2")
    print("  " + "-" * 38)
    try:
        opts = lire_arguments(argv)
        if opts["menu"]:
            return menu(opts)
        atelier = ouvrir(opts)

        voulus.update({n: True for n in opts["activer"]})
        voulus.update({n: False for n in opts["retirer"]})
        if voulus:
            faire(atelier, voulus)
        afficher(atelier)
        print()
        return 0
    except Refus as e:
        print("\n  ABORTED: %s\n" % e)
        return 2

{

    if (n == cap) {

def simuler(chemin):
    """A 15-minute game: deck, map, piloting a plane, combat, one-minute pause."""
    random.seed(4)
    f = Fichier()
    noms = ["holomap_screen", "screen_veh_m", "screen_veh_l", "screen_veh_r", "screen_nav_m", "screen_inv_l",
            "screen_inv_r", "screen_com_l", "screen_nav_l", "screen_nav_r"]
    ecrans = {n: ecran(f, n) for n in noms}
    hud = None
    e0, tick, fps = 1792850000, 0.0, 20
    iles = [(0, 2, 0, False)] * 44 + [(0, 59, 0, False)] * 9 + [(0, 77, 0, False)] * 8
    for k in range(15 * 60 * fps):
        s = k / fps
        m = s / 60
        pilote = 6.0 <= m < 9.0
        combat = 7.0 <= m < 8.5
        pause = 12.0 <= m < 13.0
        carte = 3.0 <= m < 4.5
        perdues = int(max(0.0, min(m, 8.5) - 7.0) * 6)
        unites = [(1, 0, 0, True), (1, 8, 0 if 2 <= m < 11 else 4, True), (1, 8, 0 if 2.5 <= m < 11 else 4, True),
                  (1, 12, 4, True), (1, 2, 0, True), (1, 2, 4, True), (1, 64, 0, True),
                  (2, 0, 0, False), (2, 8, 0, False), (2, 10, 0, m > 5), (2, 2, 4, False)] + iles[perdues:]
        vitesse = 0.0 if pause else max(0.35, 0.74 - 0.0015 * (len(unites) - 60) - (0.08 if pilote else 0)
                                         - (0.12 if combat else 0) + random.gauss(0, 0.01))
        tick += vitesse * 30 / fps
        e = e0 + int(s)
        if k % fps == 0:
            for n, lua in ecrans.items():
                mettre_unites(lua, unites)
                g = lua.globals()
                g.missiles = int(combat) * random.randint(4, 12)
                g.obus = int(combat) * random.randint(20, 45) + random.randint(0, 2)
                g.focus = (n == "holomap_screen" and carte)
            if pilote and hud is None:
                hud = ecran(f, "hud", genre="hud")
            if hud is not None:
                mettre_unites(hud, unites)
                hud.globals().missiles = ecrans["holomap_screen"].globals().missiles
        actifs = noms[:3] if pilote else noms        # while piloting, only 3 deck screens are still running
        for n in actifs:
            image(ecrans[n], e, int(tick))
        if pilote:
            image(hud, e, int(tick), genre="hud", vehicule_pilote=8)
    with open(chemin, "w") as sortie:
        sortie.write("\n".join(f.lignes) + "\n")
    print("\nSimulated game: %d lines written to %s" % (len(f.lignes), chemin))


tests()

    simuler(sys.argv[sys.argv.index("--fichier") + 1])
print("\nRESULT: " + ("ALL PASSED" if ok else "THERE ARE FAILURES"))
sys.exit(0 if ok else 1)
